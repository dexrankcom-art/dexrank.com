import fs from 'fs';
import path from 'path';
import matter from 'gray-matter';

export interface GuideFrontmatter {
  title: string;
  description: string;
  publishedAt: string;
  updatedAt?: string;
  author?: string;
  tags?: string[];
  readingTime?: string;
  difficulty?: 'beginner' | 'intermediate' | 'advanced';
}

export interface Guide {
  slug: string;
  frontmatter: GuideFrontmatter;
  content: string;
}

export interface GuideListItem {
  slug: string;
  title: string;
  description: string;
  publishedAt: string;
  readingTime?: string;
  difficulty?: 'beginner' | 'intermediate' | 'advanced';
}

const GUIDES_DIR = path.join(process.cwd(), 'content/guides');

/**
 * Get a single guide by slug
 */
export async function getGuideBySlug(slug: string): Promise<Guide | null> {
  const filePath = path.join(GUIDES_DIR, `${slug}.mdx`);

  if (!fs.existsSync(filePath)) {
    return null;
  }

  try {
    const source = fs.readFileSync(filePath, 'utf8');
    const { content, data } = matter(source);

    return {
      slug,
      frontmatter: data as GuideFrontmatter,
      content,
    };
  } catch {
    return null;
  }
}

/**
 * Get all guide slugs for static generation
 */
export async function getAllGuideSlugs(): Promise<string[]> {
  if (!fs.existsSync(GUIDES_DIR)) {
    return [];
  }

  const files = fs.readdirSync(GUIDES_DIR);
  return files
    .filter((file) => file.endsWith('.mdx'))
    .map((file) => file.replace(/\.mdx$/, ''));
}

/**
 * Get all guides for listing
 */
export async function getAllGuides(): Promise<GuideListItem[]> {
  const slugs = await getAllGuideSlugs();
  const guides: GuideListItem[] = [];

  for (const slug of slugs) {
    const guide = await getGuideBySlug(slug);
    if (guide) {
      guides.push({
        slug,
        title: guide.frontmatter.title,
        description: guide.frontmatter.description,
        publishedAt: guide.frontmatter.publishedAt,
        readingTime: guide.frontmatter.readingTime,
        difficulty: guide.frontmatter.difficulty,
      });
    }
  }

  // Sort by publish date (newest first)
  return guides.sort((a, b) =>
    new Date(b.publishedAt).getTime() - new Date(a.publishedAt).getTime()
  );
}

## What is a Decentralized Exchange?

A **decentralized exchange (DEX)** is a cryptocurrency trading platform that operates without a central authority. Unlike traditional exchanges like Coinbase or Binance, DEXs let you trade directly from your wallet without depositing funds to a third party.

The key difference: **you always control your crypto**. Your tokens never leave your wallet until the moment of the swap.

## How Do DEXs Work?

Most DEXs use a system called an **Automated Market Maker (AMM)**. Instead of matching buyers with sellers, AMMs use liquidity pools - pools of tokens that anyone can trade against.

Here's the simple version:

1. **Liquidity providers** deposit pairs of tokens (like ETH and USDC) into a pool
2. **Traders** swap tokens by trading against the pool
3. **Smart contracts** automatically calculate prices based on the ratio of tokens in the pool
4. **Fees** (usually 0.3%) go to liquidity providers as rewards

## Why Use a DEX?

### Advantages

- **Self-custody**: Your keys, your crypto. No exchange hacks can steal your funds.
- **No KYC**: Trade without submitting identity documents
- **Access to new tokens**: DEXs list tokens permissionlessly - no waiting for exchange approval
- **Transparency**: All trades are on-chain and verifiable
- **Composability**: DEXs integrate with other DeFi protocols

### Disadvantages

- **Gas fees**: On Ethereum, transaction costs can be high during busy periods
- **Complexity**: Requires understanding wallets, gas, and blockchain basics
- **Slippage**: Large trades can move the price significantly
- **No customer support**: You're responsible for your own transactions

## Popular DEXs by Category

**Spot Trading (AMMs)**
- Uniswap (Ethereum, Layer 2s)
- PancakeSwap (BNB Chain)
- Raydium (Solana)

**Perpetual Trading**
- Hyperliquid (HyperEVM)
- GMX (Arbitrum)
- dYdX (Cosmos)

**Aggregators**
- Jupiter (Solana)
- 1inch (Multi-chain)

## Getting Started

Ready to try a DEX? Here's what you need:

1. A crypto wallet (MetaMask for Ethereum, Phantom for Solana)
2. Some crypto to trade (and extra for gas fees)
3. A few minutes to connect your wallet and make your first swap

Start with small amounts while you learn. The beauty of DeFi is that you can experiment without asking anyone's permission.

## The Fundamental Difference

**Centralized Exchanges (CEXs)** like Coinbase, Binance, and Kraken are companies that hold your crypto for you. You deposit funds, trade on their internal system, and withdraw when needed.

**Decentralized Exchanges (DEXs)** like Uniswap, PancakeSwap, and Jupiter never hold your funds. You trade directly from your wallet using smart contracts.

## Side-by-Side Comparison

| Factor | CEX | DEX |
|--------|-----|-----|
| **Custody** | Exchange holds your crypto | You hold your crypto |
| **KYC Required** | Yes | No |
| **Account Recovery** | Customer support can help | Only you have access |
| **Hack Risk** | Exchange can be hacked | Smart contract risk only |
| **Trading Fees** | 0.1-0.5% typically | 0.05-0.3% + gas |
| **Token Selection** | Curated listing | Permissionless |
| **Fiat On-ramp** | Yes | Limited |
| **Speed** | Instant | Depends on blockchain |

## When to Use a CEX

Choose a centralized exchange when:

- **You're new to crypto** - Simpler interface, customer support available
- **You need fiat on/off-ramps** - Buy crypto with your bank account
- **You're trading large volumes** - Better liquidity for major pairs
- **You want margin/futures** - More sophisticated trading products
- **You need account recovery** - Lost password? CEXs can help

## When to Use a DEX

Choose a decentralized exchange when:

- **Security is your priority** - Self-custody means no exchange hacks
- **You value privacy** - No KYC requirements
- **You want new tokens** - Access tokens before CEX listings
- **You use DeFi** - Composable with other protocols
- **You distrust intermediaries** - True peer-to-peer trading

## The Security Tradeoff

**CEX Risk**: If Binance or Coinbase gets hacked, your funds could be lost. History shows this happens - Mt. Gox, FTX, and others have lost user funds.

**DEX Risk**: If you lose your wallet seed phrase, nobody can help. Smart contracts can have bugs, though major DEXs are extensively audited.

Neither is "safer" - they have different risk profiles.

## Cost Comparison

**CEX Costs**
- Trading fee: ~0.1-0.5%
- Withdrawal fee: Fixed amount per coin
- Spread: Usually tight for major pairs

**DEX Costs**
- Trading fee: ~0.05-0.3%
- Gas fee: Varies by blockchain (high on Ethereum, low on Solana/L2s)
- Slippage: Can be significant for large trades

## Our Recommendation

**Start with both**. Use a CEX for fiat on-ramps and your first trades. As you get comfortable, start using DEXs for better privacy, new tokens, and true ownership of your assets.

The future is hybrid - even experienced traders use both for different purposes.

## What is a Liquidity Pool?

A **liquidity pool** is a collection of tokens locked in a smart contract. Instead of matching buyers with sellers, DEXs let traders swap against these pools.

Think of it like a vending machine: it always has inventory ready, and you don't need to wait for someone to sell you what you want.

## How Liquidity Pools Work

### The Basic Formula

Most pools use the **constant product formula**:

```
x * y = k
```

Where:
- `x` = amount of Token A
- `y` = amount of Token B
- `k` = constant (the product must stay the same)

When you swap ETH for USDC, you're adding ETH to the pool and removing USDC. The formula ensures there's always some of both tokens available.

### Why Prices Change

As the ratio of tokens in the pool changes, so does the price. This is automatic - no order books, no market makers, just math.

**Example**: A pool with 10 ETH and 10,000 USDC prices ETH at 1,000 USDC. If someone buys 1 ETH, the pool now has 9 ETH and ~11,111 USDC, making the new price ~1,234 USDC per ETH.

## Becoming a Liquidity Provider

### How to Add Liquidity

1. Choose a pool (e.g., ETH/USDC on Uniswap)
2. Deposit equal value of both tokens (e.g., $500 ETH + $500 USDC)
3. Receive LP tokens representing your share of the pool
4. Earn trading fees proportional to your share

### What You Earn

- **Trading fees**: Usually 0.3% of each swap, split among all LPs
- **Token incentives**: Some pools offer additional rewards (yield farming)
- **Your share grows**: As fees accumulate, your position is worth more

### The Risks

- **Impermanent loss**: If token prices diverge, you may end up with less value than holding
- **Smart contract risk**: Bugs could drain the pool
- **Rug pulls**: For new tokens, the team could drain liquidity

## Types of Liquidity Pools

### Constant Product (Uniswap v2 style)
- Simple 50/50 split
- Good for most token pairs
- Higher slippage for large trades

### Concentrated Liquidity (Uniswap v3)
- LPs choose a price range
- Better capital efficiency
- More active management required

### Stable Pools (Curve style)
- Optimized for pegged assets (USDC/USDT)
- Very low slippage
- Lower fees

## Is LP Worth It?

**Good for LPs**:
- Stable pairs with high volume (ETH/USDC, BTC/ETH)
- Pools with additional incentives
- Pairs you'd hold anyway

**Risky for LPs**:
- Volatile new tokens
- Low-volume pairs
- Anything with major price movement expected

Start small, understand the risks, and calculate your actual returns including impermanent loss.

## What is Impermanent Loss?

**Impermanent loss (IL)** is the difference between holding tokens in a liquidity pool versus simply holding them in your wallet. It occurs when the price ratio of your deposited tokens changes.

The "impermanent" part: if prices return to your entry point, the loss disappears. If you withdraw while prices are different, the loss becomes permanent.

## How Impermanent Loss Works

### Simple Example

You deposit 1 ETH ($1,000) + 1,000 USDC into a pool.
Total value: $2,000

**Scenario**: ETH doubles to $2,000

**If you had just held**:
- 1 ETH = $2,000
- 1,000 USDC = $1,000
- Total: $3,000

**In the liquidity pool** (due to rebalancing):
- ~0.707 ETH = $1,414
- ~1,414 USDC = $1,414
- Total: $2,828

**Impermanent loss**: $3,000 - $2,828 = $172 (5.7%)

## The IL Formula

For a 50/50 pool, impermanent loss depends only on the price change ratio:

| Price Change | Impermanent Loss |
|--------------|------------------|
| 1.25x (25% up) | 0.6% |
| 1.50x (50% up) | 2.0% |
| 2x (100% up) | 5.7% |
| 3x (200% up) | 13.4% |
| 5x (400% up) | 25.5% |

**Key insight**: IL happens in both directions. ETH going up OR down causes IL. Only the magnitude matters, not the direction.

## When IL is Worth It

**IL can be offset by**:
- Trading fees (0.3% per swap adds up)
- Farming rewards (additional token incentives)
- High volume (more fees = more compensation)

**Rule of thumb**: If fee APY > potential IL, providing liquidity makes sense.

## Strategies to Minimize IL

### 1. Choose Correlated Pairs
- ETH/stETH (both track ETH)
- USDC/USDT (both stable)
- WBTC/renBTC (both track BTC)

### 2. Use Concentrated Liquidity Wisely
Uniswap v3 lets you concentrate liquidity in a price range. This amplifies both fees AND impermanent loss.

### 3. Time Your Exit
Monitor price ratios. If they've moved significantly and you expect further movement, consider exiting before more IL accumulates.

### 4. Factor IL Into Returns
Don't just look at APY - calculate net returns after IL. Many "high APY" pools are unprofitable after IL.

Impermanent loss isn't always bad - it's a tradeoff. You're exchanging potential upside (if one token moons) for steady fee income.

For volatile pairs: expect significant IL
For stable pairs: IL is minimal
For everything in between: do the math

## What is an Automated Market Maker?

An **Automated Market Maker (AMM)** is a type of decentralized exchange protocol that uses mathematical formulas to price assets instead of using an order book.

Traditional exchanges match buyers and sellers. AMMs let users trade against a pool of tokens, with prices determined algorithmically.

## The Evolution of DEX Design

### Order Book DEXs (2017-2018)
Early DEXs like EtherDelta tried to replicate centralized exchange order books on-chain. Problem: on-chain order matching is slow and expensive.

### AMMs (2018-present)
Bancor introduced the AMM concept. Uniswap simplified it and achieved product-market fit. Now AMMs dominate DEX trading.

## How AMMs Calculate Prices

### Constant Product Market Maker (CPMM)

Used by: Uniswap v2, SushiSwap, PancakeSwap

Formula: `x * y = k`

Where x and y are token reserves, k is a constant.

**How swaps work**:
1. Trader sends Token A to pool
2. Pool calculates new reserves maintaining k
3. Pool sends appropriate amount of Token B to trader
4. New price = new ratio of x to y

**Pros**: Simple, always has liquidity at any price
**Cons**: Capital inefficient, high slippage for large trades

### Constant Sum Market Maker (CSMM)

Formula: `x + y = k`

Creates zero slippage trades at a fixed price. Problem: pool can be drained entirely of one asset. Not used in practice for general trading.

### StableSwap (Curve)

Formula: Complex hybrid between constant product and constant sum

Optimized for assets that should trade 1:1 (stablecoins, wrapped assets). Provides very low slippage around the expected price.

**Pros**: Extremely low slippage for stable pairs
**Cons**: Only works for correlated assets

### Concentrated Liquidity (Uniswap v3)

LPs provide liquidity within specific price ranges. Mathematically equivalent to amplifying constant product within that range.

**Pros**: 4000x more capital efficient for tight ranges
**Cons**: Requires active management, IL amplified

## The Math Behind Swaps

For a constant product AMM with reserves (x, y):

**Swap calculation** (trading dx of Token X for dy of Token Y):

```
(x + dx) * (y - dy) = k
dy = y - k/(x + dx)
dy = y * dx / (x + dx)  [simplified]
```

**Price** (instantaneous):
```
price = y/x
```

**Price impact** (slippage):
```
price_impact = dx / (x + dx)
```

## Why AMMs Work

### Arbitrage Keeps Prices Correct

If an AMM price deviates from the "true" market price, arbitrageurs profit by trading until prices align. This happens automatically and near-instantly.

### LPs Are Compensated

Liquidity providers earn fees (typically 0.3%) on every trade. This compensates them for impermanent loss and opportunity cost.

### No Counterparty Needed

Unlike order books, traders don't need someone to take the other side. The pool is always ready to trade.

## AMM Limitations

### Capital Inefficiency

Most liquidity sits unused. In a standard AMM, capital is spread across all possible prices from 0 to infinity.

### Impermanent Loss

LPs are essentially selling the better-performing asset and buying the worse-performing one - continuously.

### MEV Vulnerability

On-chain trades are visible before execution. Bots can front-run or sandwich transactions for profit at traders' expense.

## Future of AMMs

### Dynamic Fees
Adjust fees based on volatility to better compensate LPs during high-risk periods.

### MEV-Resistant Designs
Batch auctions, encrypted mempools, and MEV redistribution to LPs.

### Intent-Based Systems
CoW Protocol and similar systems match user intents off-chain, reducing MEV and improving prices.
