import { db } from '@/db';
import {
  protocols,
  protocolMetrics,
  protocolChains,
  chains,
  syncStatus,
} from '@/db/schema';
import { eq, desc, sql } from 'drizzle-orm';

export type DexWithMetrics = {
  id: number;
  slug: string;
  name: string;
  symbol: string | null;
  category: string | null;
  logo: string | null;
  url: string | null;
  description: string | null;
  chains: string[];
  metrics: {
    tvl: number | null;
    tvlChange1d: number | null;
    tvlChange7d: number | null;
    volume24h: number | null;
    volume7d: number | null;
    volume30d: number | null;
    volumeChange1d: number | null;
    fetchedAt: Date;
  } | null;
};

/**
 * Get all DEXes with their latest metrics
 */
export async function getAllDexes(): Promise<DexWithMetrics[]> {
  // Get all protocols with their latest metrics using a subquery
  const result = await db
    .select({
      id: protocols.id,
      slug: protocols.slug,
      name: protocols.name,
      symbol: protocols.symbol,
      category: protocols.category,
      logo: protocols.logo,
      url: protocols.url,
      description: protocols.description,
    })
    .from(protocols)
    .orderBy(protocols.name);

  // For each protocol, get chains and latest metrics
  const dexes: DexWithMetrics[] = [];

  for (const proto of result) {
    // Get chains for this protocol
    const chainResults = await db
      .select({ name: chains.name })
      .from(protocolChains)
      .innerJoin(chains, eq(protocolChains.chainId, chains.id))
      .where(eq(protocolChains.protocolId, proto.id));

    // Get latest metrics
    const [latestMetrics] = await db
      .select({
        tvl: protocolMetrics.tvl,
        tvlChange1d: protocolMetrics.tvlChange1d,
        tvlChange7d: protocolMetrics.tvlChange7d,
        volume24h: protocolMetrics.volume24h,
        volume7d: protocolMetrics.volume7d,
        volume30d: protocolMetrics.volume30d,
        volumeChange1d: protocolMetrics.volumeChange1d,
        fetchedAt: protocolMetrics.fetchedAt,
      })
      .from(protocolMetrics)
      .where(eq(protocolMetrics.protocolId, proto.id))
      .orderBy(desc(protocolMetrics.fetchedAt))
      .limit(1);

    dexes.push({
      ...proto,
      chains: chainResults.map((c) => c.name),
      metrics: latestMetrics || null,
    });
  }

  return dexes;
}

/**
 * Get a single DEX by slug with full details
 */
export async function getDexBySlug(slug: string): Promise<DexWithMetrics | null> {
  const [proto] = await db
    .select()
    .from(protocols)
    .where(eq(protocols.slug, slug))
    .limit(1);

  if (!proto) {
    return null;
  }

  // Get chains
  const chainResults = await db
    .select({ name: chains.name })
    .from(protocolChains)
    .innerJoin(chains, eq(protocolChains.chainId, chains.id))
    .where(eq(protocolChains.protocolId, proto.id));

  // Get latest metrics
  const [latestMetrics] = await db
    .select({
      tvl: protocolMetrics.tvl,
      tvlChange1d: protocolMetrics.tvlChange1d,
      tvlChange7d: protocolMetrics.tvlChange7d,
      volume24h: protocolMetrics.volume24h,
      volume7d: protocolMetrics.volume7d,
      volume30d: protocolMetrics.volume30d,
      volumeChange1d: protocolMetrics.volumeChange1d,
      fetchedAt: protocolMetrics.fetchedAt,
    })
    .from(protocolMetrics)
    .where(eq(protocolMetrics.protocolId, proto.id))
    .orderBy(desc(protocolMetrics.fetchedAt))
    .limit(1);

  return {
    id: proto.id,
    slug: proto.slug,
    name: proto.name,
    symbol: proto.symbol,
    category: proto.category,
    logo: proto.logo,
    url: proto.url,
    description: proto.description,
    chains: chainResults.map((c) => c.name),
    metrics: latestMetrics || null,
  };
}

/**
 * Get sync status to check data freshness
 */
export async function getSyncStatus(): Promise<{
  lastSync: Date | null;
  isStale: boolean;
  hoursOld: number | null;
}> {
  const [status] = await db
    .select()
    .from(syncStatus)
    .where(eq(syncStatus.syncType, 'full'))
    .limit(1);

  if (!status?.lastSuccessAt) {
    return { lastSync: null, isStale: true, hoursOld: null };
  }

  const hoursOld = (Date.now() - status.lastSuccessAt.getTime()) / (1000 * 60 * 60);

  return {
    lastSync: status.lastSuccessAt,
    isStale: hoursOld > 12, // Data is stale if >12 hours old
    hoursOld: Math.round(hoursOld * 10) / 10,
  };
}
```

Note: This implementation uses multiple queries per DEX for clarity. In production with 100+ DEXs, this could be optimized with a single aggregated query, but the current approach is readable and correct for v1.

/**
 * Simple in-memory cache for fallback when database is unavailable.
 * In serverless, this is per-instance - good enough for graceful degradation.
 *
 * For true distributed caching, use Vercel KV or Upstash Redis in production.
 */

type CacheEntry<T> = {
  data: T;
  timestamp: number;
  ttl: number;
};

const cache = new Map<string, CacheEntry<unknown>>();

const DEFAULT_TTL = 5 * 60 * 1000; // 5 minutes

/**
 * Get cached data if available and not expired
 */
export function getCachedData<T>(key: string): T | null {
  const entry = cache.get(key) as CacheEntry<T> | undefined;

  if (!entry) {
    return null;
  }

  const isExpired = Date.now() - entry.timestamp > entry.ttl;

  if (isExpired) {
    cache.delete(key);
    return null;
  }

  return entry.data;
}

/**
 * Store data in cache
 */
export function setCachedData<T>(key: string, data: T, ttl = DEFAULT_TTL): void {
  cache.set(key, {
    data,
    timestamp: Date.now(),
    ttl,
  });
}

/**
 * Higher-order function that wraps a data fetcher with cache fallback.
 * - On success: returns fresh data, updates cache
 * - On failure: returns cached data if available, otherwise throws
 */
export async function withCacheFallback<T>(
  key: string,
  fetcher: () => Promise<T>,
  options: { ttl?: number; staleWhileRevalidate?: boolean } = {}
): Promise<{ data: T; fromCache: boolean; error?: string }> {
  const { ttl = DEFAULT_TTL, staleWhileRevalidate = true } = options;

  try {
    // Try to fetch fresh data
    const data = await fetcher();

    // Update cache with fresh data
    setCachedData(key, data, ttl);

    return { data, fromCache: false };
  } catch (error) {
    const errorMessage = error instanceof Error ? error.message : String(error);
    console.error(`Fetch failed for ${key}:`, errorMessage);

    // Try to return cached data
    const cachedData = getCachedData<T>(key);

    if (cachedData !== null) {
      console.log(`Returning cached data for ${key}`);
      return {
        data: cachedData,
        fromCache: true,
        error: `Using cached data due to: ${errorMessage}`,
      };
    }

    // No cache available, rethrow
    throw error;
  }
}

/**
 * Clear all cached data (useful for testing)
 */
export function clearCache(): void {
  cache.clear();
}

/**
 * Get cache stats (useful for monitoring)
 */
export function getCacheStats(): { size: number; keys: string[] } {
  return {
    size: cache.size,
    keys: Array.from(cache.keys()),
  };
}
```

Key design decisions:
- In-memory cache is simple and works per-serverless-instance
- TTL defaults to 5 minutes (matches typical API request patterns)
- `withCacheFallback` pattern makes it easy to wrap any async function
- Returns metadata about whether data came from cache
- Logs when falling back to cache for monitoring

import { NextRequest } from 'next/server';
import { getAllDexes, getSyncStatus, type DexWithMetrics } from '@/lib/data/dex';
import { withCacheFallback } from '@/lib/data/cache';

export const dynamic = 'force-dynamic';

type DexListResponse = {
  dexes: DexWithMetrics[];
  count: number;
  syncStatus: {
    lastSync: string | null;
    isStale: boolean;
    hoursOld: number | null;
  };
  meta: {
    fromCache: boolean;
    warning?: string;
  };
};

export async function GET(request: NextRequest) {
  try {
    const { data: dexes, fromCache, error } = await withCacheFallback(
      'all-dexes',
      getAllDexes,
      { ttl: 5 * 60 * 1000 } // 5 minute cache
    );

    const status = await getSyncStatus();

    const response: DexListResponse = {
      dexes,
      count: dexes.length,
      syncStatus: {
        lastSync: status.lastSync?.toISOString() || null,
        isStale: status.isStale,
        hoursOld: status.hoursOld,
      },
      meta: {
        fromCache,
      },
    };

    // Add warnings if applicable
    if (fromCache && error) {
      response.meta.warning = error;
    } else if (status.isStale) {
      response.meta.warning = `Data is ${status.hoursOld} hours old. Sync may have failed.`;
    }

    return Response.json(response);
  } catch (error) {
    console.error('Failed to fetch DEXes:', error);
    return Response.json(
      {
        error: 'Failed to fetch DEX data',
        message: error instanceof Error ? error.message : 'Unknown error',
      },
      { status: 500 }
    );
  }
}

import { NextRequest } from 'next/server';
import { getDexBySlug, type DexWithMetrics } from '@/lib/data/dex';
import { withCacheFallback } from '@/lib/data/cache';

export const dynamic = 'force-dynamic';

type DexResponse = {
  dex: DexWithMetrics;
  meta: {
    fromCache: boolean;
    warning?: string;
  };
};

export async function GET(
  request: NextRequest,
  { params }: { params: Promise<{ slug: string }> }
) {
  const { slug } = await params;

  try {
    const { data: dex, fromCache, error } = await withCacheFallback(
      `dex-${slug}`,
      () => getDexBySlug(slug),
      { ttl: 5 * 60 * 1000 }
    );

    if (!dex) {
      return Response.json(
        { error: 'DEX not found', slug },
        { status: 404 }
      );
    }

    const response: DexResponse = {
      dex,
      meta: {
        fromCache,
      },
    };

    if (fromCache && error) {
      response.meta.warning = error;
    }

    return Response.json(response);
  } catch (error) {
    console.error(`Failed to fetch DEX ${slug}:`, error);
    return Response.json(
      {
        error: 'Failed to fetch DEX data',
        slug,
        message: error instanceof Error ? error.message : 'Unknown error',
      },
      { status: 500 }
    );
  }
}
