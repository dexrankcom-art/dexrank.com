```typescript
import { backOff } from 'exponential-backoff';
import {
  ProtocolSchema,
  DexVolumeResponseSchema,
  type Protocol,
  type DexVolumeResponse,
} from './types';

const BASE_URL = 'https://api.llama.fi';

async function fetchWithRetry<T>(url: string): Promise<T> {
  return backOff(
    async () => {
      const response = await fetch(url, {
        headers: { Accept: 'application/json' },
        next: { revalidate: 0 }, // No caching for sync
      });

      if (!response.ok) {
        throw new Error(`HTTP ${response.status}: ${response.statusText}`);
      }

      return response.json();
    },
    {
      numOfAttempts: 3,
      startingDelay: 1000,
      maxDelay: 10000,
      jitter: 'full',
      retry: (error, attemptNumber) => {
        console.warn(`DefiLlama API attempt ${attemptNumber} failed:`, error.message);
        const status = error.message.match(/HTTP (\d+)/)?.[1];
        // Retry on rate limit (429) and server errors (5xx)
        return !status || status === '429' || parseInt(status) >= 500;
      },
    }
  );
}

/**
 * Fetch all protocols from DefiLlama /protocols endpoint
 * Filters to DEX category only
 */
export async function fetchProtocols(): Promise<Protocol[]> {
  console.log('Fetching protocols from DefiLlama...');
  const data = await fetchWithRetry<unknown[]>(`${BASE_URL}/protocols`);

  // Parse and filter to DEXes only
  const protocols: Protocol[] = [];
  for (const item of data) {
    const parsed = ProtocolSchema.safeParse(item);
    if (parsed.success && parsed.data.category === 'Dexes') {
      protocols.push(parsed.data);
    }
  }

  console.log(`Fetched ${protocols.length} DEX protocols`);
  return protocols;
}

/**
 * Fetch DEX volume data from DefiLlama /overview/dexs endpoint
 */
export async function fetchDexVolumes(): Promise<DexVolumeResponse> {
  console.log('Fetching DEX volumes from DefiLlama...');
  const data = await fetchWithRetry<unknown>(`${BASE_URL}/overview/dexs`);
  const parsed = DexVolumeResponseSchema.parse(data);
  console.log(`Fetched volume data for ${parsed.protocols.length} DEXes`);
  return parsed;
}
```

Key implementation notes:
- Use Zod `.safeParse()` for protocols to skip malformed entries rather than failing entirely
- Filter to "Dexes" category to get only DEX protocols (not all DeFi protocols)
- Exponential backoff with jitter prevents thundering herd on retries
- Log attempt failures for debugging

Create `src/lib/sync/protocols.ts`:

```typescript
import { db } from '@/db';
import { protocols, protocolChains, protocolMetrics, chains } from '@/db/schema';
import { fetchProtocols } from '@/lib/defillama/client';
import { eq, sql } from 'drizzle-orm';

export async function syncProtocols(): Promise<{ count: number }> {
  console.log('Starting protocol sync...');

  const apiProtocols = await fetchProtocols();

  // Get existing chains for mapping
  const existingChains = await db.select().from(chains);
  const chainMap = new Map(existingChains.map((c) => [c.slug.toLowerCase(), c.id]));

  let count = 0;

  for (const proto of apiProtocols) {
    // Upsert protocol
    const [upsertedProtocol] = await db
      .insert(protocols)
      .values({
        defillamaId: proto.id,
        slug: proto.slug,
        name: proto.name,
        symbol: proto.symbol ?? null,
        category: proto.category ?? 'Dexes',
        logo: proto.logo ?? null,
        url: proto.url ?? null,
      })
      .onConflictDoUpdate({
        target: protocols.defillamaId,
        set: {
          slug: proto.slug,
          name: proto.name,
          symbol: proto.symbol ?? null,
          logo: proto.logo ?? null,
          url: proto.url ?? null,
          updatedAt: new Date(),
        },
      })
      .returning({ id: protocols.id });

    const protocolId = upsertedProtocol.id;

    // Upsert protocol-chain associations
    for (const chainName of proto.chains) {
      const chainId = chainMap.get(chainName.toLowerCase());
      if (chainId) {
        await db
          .insert(protocolChains)
          .values({ protocolId, chainId })
          .onConflictDoNothing();
      }
    }

    // Insert latest metrics
    await db.insert(protocolMetrics).values({
      protocolId,
      tvl: proto.tvl,
      tvlChange1h: proto.change_1h ?? null,
      tvlChange1d: proto.change_1d ?? null,
      tvlChange7d: proto.change_7d ?? null,
      chainTvls: proto.chainTvls ?? null,
      fetchedAt: new Date(),
    });

    count++;
  }

  console.log(`Synced ${count} protocols`);
  return { count };
}
```

Create `src/lib/sync/volumes.ts`:

```typescript
import { db } from '@/db';
import { protocols, protocolMetrics } from '@/db/schema';
import { fetchDexVolumes } from '@/lib/defillama/client';
import { eq, desc } from 'drizzle-orm';

export async function syncVolumes(): Promise<{ count: number }> {
  console.log('Starting volume sync...');

  const volumeData = await fetchDexVolumes();

  // Get existing protocols for matching
  const existingProtocols = await db.select().from(protocols);
  const protocolMap = new Map(
    existingProtocols.map((p) => [p.name.toLowerCase(), p.id])
  );

  let count = 0;

  for (const dex of volumeData.protocols) {
    // Match by name (DefiLlama uses different identifiers for volume vs TVL)
    const protocolId = protocolMap.get(dex.name.toLowerCase());

    if (!protocolId) {
      // Protocol not in our DB yet, skip (will be added in next protocol sync)
      continue;
    }

    // Get the most recent metrics for this protocol to update
    const [latestMetric] = await db
      .select()
      .from(protocolMetrics)
      .where(eq(protocolMetrics.protocolId, protocolId))
      .orderBy(desc(protocolMetrics.fetchedAt))
      .limit(1);

    if (latestMetric) {
      // Update existing metric with volume data
      await db
        .update(protocolMetrics)
        .set({
          volume24h: dex.total24h,
          volume7d: dex.total7d,
          volume30d: dex.total30d,
          volumeChange1d: dex.change_1d ?? null,
          volumeChange7d: dex.change_7d ?? null,
          updatedAt: new Date(),
        })
        .where(eq(protocolMetrics.id, latestMetric.id));
    } else {
      // No metrics exist yet, create one with volume data only
      await db.insert(protocolMetrics).values({
        protocolId,
        volume24h: dex.total24h,
        volume7d: dex.total7d,
        volume30d: dex.total30d,
        volumeChange1d: dex.change_1d ?? null,
        volumeChange7d: dex.change_7d ?? null,
        fetchedAt: new Date(),
      });
    }

    count++;
  }

  console.log(`Synced volumes for ${count} DEXes`);
  return { count };
}
```

```typescript
export { syncProtocols } from './protocols';
export { syncVolumes } from './volumes';

import { db } from '@/db';
import { syncStatus } from '@/db/schema';
import { syncProtocols } from './protocols';
import { syncVolumes } from './volumes';

/**
 * Run full sync: protocols first, then volumes
 * Protocols must run first to establish the protocol records that volumes reference
 */
export async function syncAll(): Promise<{
  protocols: { count: number };
  volumes: { count: number };
}> {
  console.log('Starting full sync...');

  // Sync protocols first (creates/updates protocol records)
  const protocolResult = await syncProtocols();

  // Small delay to avoid overwhelming the API
  await new Promise((resolve) => setTimeout(resolve, 500));

  // Sync volumes (updates existing protocol metrics)
  const volumeResult = await syncVolumes();

  // Update sync status
  await db
    .insert(syncStatus)
    .values({
      syncType: 'full',
      lastSyncAt: new Date(),
      lastSuccessAt: new Date(),
      recordsProcessed: protocolResult.count + volumeResult.count,
    })
    .onConflictDoUpdate({
      target: syncStatus.syncType,
      set: {
        lastSyncAt: new Date(),
        lastSuccessAt: new Date(),
        recordsProcessed: protocolResult.count + volumeResult.count,
        lastError: null,
        updatedAt: new Date(),
      },
    });

  console.log('Full sync complete');
  return { protocols: protocolResult, volumes: volumeResult };
}
```

Key implementation notes:
- Protocols sync first because volumes need protocol IDs to update
- Use upsert (onConflictDoUpdate) to handle re-runs safely
- Match volumes to protocols by name (lowercase) since DefiLlama uses different IDs
- Skip unknown protocols in volume sync - they'll be picked up in next protocol sync

```typescript
import { NextRequest } from 'next/server';
import { db } from '@/db';
import { syncStatus } from '@/db/schema';
import { syncAll } from '@/lib/sync';

// Allow up to 60 seconds for sync to complete
export const maxDuration = 60;
export const dynamic = 'force-dynamic';

export async function GET(request: NextRequest) {
  // Verify cron secret (Vercel sends this header)
  const authHeader = request.headers.get('authorization');
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    console.warn('Unauthorized cron request');
    return new Response('Unauthorized', { status: 401 });
  }

  const startTime = Date.now();

  try {
    const result = await syncAll();

    return Response.json({
      success: true,
      duration: Date.now() - startTime,
      protocols: result.protocols.count,
      volumes: result.volumes.count,
    });
  } catch (error) {
    console.error('Sync failed:', error);

    // Record failure in sync status
    await db
      .insert(syncStatus)
      .values({
        syncType: 'full',
        lastSyncAt: new Date(),
        lastError: error instanceof Error ? error.message : String(error),
      })
      .onConflictDoUpdate({
        target: syncStatus.syncType,
        set: {
          lastSyncAt: new Date(),
          lastError: error instanceof Error ? error.message : String(error),
          updatedAt: new Date(),
        },
      });

    return Response.json(
      {
        success: false,
        duration: Date.now() - startTime,
        error: error instanceof Error ? error.message : String(error),
      },
      { status: 500 }
    );
  }
}
```
